Weighted Exam Readiness & Category Mastery Engine

lib/attempts.js

Computes cumulative Category Mastery, Chapter Mastery, and the true Weighted Curriculum Readiness index based on CSC exam coverage formulas: \(\text{Readiness} = \sum (\text{Mastery}_i \times \text{Weight}_i)\).

lib/attempts.js (Core Analytics Functions) 
/**
 * Computes Dashboard Summary, Subject Performance, Chapter Mastery, and Weak Areas.
 * Professional and Sub-Professional levels are kept strictly separated.
 */
export async function getDashboardSummary(selectedLevel = "professional") {
  const normLevel = selectedLevel === "sub" ? "sub" : "professional";
  const allAttempts = await getAttemptHistory();

  // Filter attempts strictly for the active exam level
  const attempts = allAttempts.filter(
    (attempt) => attempt.level === normLevel
  );

  const curriculumCategories = getCurriculum(normLevel);
  const categoryMap = {};

  curriculumCategories.forEach((cat) => {
    categoryMap[cat.title] = {
      title: cat.title,
      estimatedWeight: cat.estimatedWeight,
      correct: 0,
      wrong: 0,
      total: 0,
      chapters: {},
    };

    cat.chapters.forEach((ch) => {
      categoryMap[cat.title].chapters[ch.title] = {
        title: ch.title,
        correct: 0,
        wrong: 0,
        total: 0,
      };
    });
  });

  // Aggregate questions across all attempts for this level
  attempts.forEach((attempt) => {
    const breakdown = attempt.subject_breakdown || [];
    breakdown.forEach((subj) => {
      if (!subj?.title) return;
      const matchedCat = Object.keys(categoryMap).find(
        (t) => t.toLowerCase() === subj.title.trim().toLowerCase()
      );
      if (!matchedCat) return;

      const catObj = categoryMap[matchedCat];
      catObj.correct += Number(subj.correct ?? 0);
      catObj.wrong += Number(subj.wrong ?? 0);
      catObj.total += Number(subj.total ?? 0);

      if (Array.isArray(subj.chapters)) {
        subj.chapters.forEach((ch) => {
          if (!ch?.title) return;
          const matchedChap = Object.keys(catObj.chapters).find(
            (t) => t.toLowerCase() === ch.title.trim().toLowerCase()
          );
          if (matchedChap) {
            catObj.chapters[matchedChap].correct += Number(ch.correct ?? 0);
            catObj.chapters[matchedChap].wrong += Number(ch.wrong ?? 0);
            catObj.chapters[matchedChap].total += Number(ch.total ?? 0);
          }
        });
      }
    });
  });

  // Calculate Category and Chapter Mastery Percentages
  const subjectReadiness = curriculumCategories.map((cat) => {
    const catData = categoryMap[cat.title];
    const catPercent =
      catData.total > 0
        ? Math.round((catData.correct / catData.total) * 100)
        : null;

    const chapters = cat.chapters.map((ch) => {
      const chData = catData.chapters[ch.title];
      const chPercent =
        chData.total > 0
          ? Math.round((chData.correct / chData.total) * 100)
          : null;
      return {
        title: ch.title,
        percent: chPercent,
        totalAttempted: chData.total,
        totalCorrect: chData.correct,
      };
    });

    return {
      title: cat.title,
      estimatedWeight: cat.estimatedWeight,
      percent: catPercent,
      totalAttempted: catData.total,
      totalCorrect: catData.correct,
      chapters,
    };
  });

  // Calculate True Weighted Curriculum Readiness:
  // Each category contributes according to its official exam weight:
  // e.g. Analytical 35%, Numerical 30%, Verbal 30%, General Information 5%.
  let weightedReadiness = 0;
  subjectReadiness.forEach((cat) => {
    const weight = Number(cat.estimatedWeight || 0);
    const scorePercent = cat.percent !== null ? Number(cat.percent || 0) : 0;
    weightedReadiness += scorePercent * weight;
  });

  const examReadiness = Math.min(100, Math.round(weightedReadiness));

  return {
    examsTaken: attempts.length,
    examReadiness,
    subjectReadiness,
  };
}

Stratified Question Sampler & Limiter Engine

lib/examData.js

Performs stratified proportional sampling across all subject pools when building full mock exams or filtered multi-category practice sets, enforcing exact item quotas with Fisher-Yates randomization.

lib/examData.js (Sampling Algorithms) 
/**
 * Samples questions proportionally across curriculum categories to match target exam limits.
 */
function sampleExamQuestions(questions, level = "professional", targetLimit = 170) {
  if (!questions || questions.length <= targetLimit) {
    return shuffleArray(questions || []);
  }

  const normLevel = level === "sub" ? "sub" : "professional";
  const curriculumCats = getCurriculum(normLevel);

  // Group questions by category
  const poolByCat = {};
  curriculumCats.forEach((c) => {
    poolByCat[c.title.toLowerCase()] = [];
  });
  const unclassifiedPool = [];

  questions.forEach((q) => {
    const key = (q.subjectTitle || "").trim().toLowerCase();
    const matchedKey = Object.keys(poolByCat).find(
      (k) => k === key || key.includes(k) || k.includes(key)
    );
    if (matchedKey) {
      poolByCat[matchedKey].push(q);
    } else {
      unclassifiedPool.push(q);
    }
  });

  const selectedQuestions = [];
  const remainingQuestions = [...unclassifiedPool];

  // Sample each category proportionally according to its curriculum weight
  curriculumCats.forEach((cat) => {
    const catKey = cat.title.toLowerCase();
    const catPool = shuffleArray(poolByCat[catKey] || []);
    const weight = cat.estimatedWeight || 1 / curriculumCats.length;
    const catQuota = Math.round(targetLimit * weight);

    const picked = catPool.slice(0, catQuota);
    const leftOver = catPool.slice(catQuota);

    selectedQuestions.push(...picked);
    remainingQuestions.push(...leftOver);
  });

  // Backfill if rounding deficit occurs
  if (selectedQuestions.length < targetLimit && remainingQuestions.length > 0) {
    const deficit = targetLimit - selectedQuestions.length;
    const shuffledRemaining = shuffleArray(remainingQuestions);
    selectedQuestions.push(...shuffledRemaining.slice(0, deficit));
  }

  const finalSet =
    selectedQuestions.length > targetLimit
      ? selectedQuestions.slice(0, targetLimit)
      : selectedQuestions;

  // Final Fisher-Yates shuffle across mixed subjects
  return shuffleArray(finalSet);
}

export function shuffleArray(arr) {
  const copy = [...arr];
  for (let i = copy.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [copy[i], copy[j]] = [copy[j], copy[i]];
  }
  return copy;
}

export function getEstimatedMinutes(count) {
  if (!count || count <= 0) return 0;
  if (count <= 10) return 15;
  if (count <= 15) return 20;
  if (count <= 20) return 25;
  if (count <= 25) return 35;
  if (count <= 30) return 40;
  if (count <= 40) return 50;
  if (count <= 50) return 60;
  return Math.round(count * 1.3);
}
